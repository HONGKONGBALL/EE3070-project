cmake_minimum_required(VERSION 3.16)
project(navigation_parser_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_subdirectory(../../firmware/components/route_parser route_parser)
add_subdirectory(../../firmware/components/mcp_client mcp_client)
add_executable(parser_tests parser_tests.cpp)
target_link_libraries(parser_tests PRIVATE route_parser)
target_compile_options(parser_tests PRIVATE -Wall -Wextra -Werror)
foreach(case_id IN ITEMS TC20 TC21 TC22 parser_rpc parser_geometry)
    add_test(NAME ${case_id} COMMAND parser_tests ${case_id})
endforeach()
add_executable(client_tests client_tests.cpp)
target_link_libraries(client_tests PRIVATE mcp_client)
add_test(NAME TC18 COMMAND client_tests)
add_executable(schema_tests schema_tests.cpp)
target_link_libraries(schema_tests PRIVATE mcp_client)
target_include_directories(schema_tests PRIVATE ../../firmware/components/mcp_client
    ../../firmware/components/route_parser ../../firmware/components/route_parser/third_party)
add_test(NAME TC43 COMMAND schema_tests)
add_executable(geometry_fixture_tests geometry_fixture_tests.cpp)
target_link_libraries(geometry_fixture_tests PRIVATE route_parser)
target_compile_definitions(geometry_fixture_tests PRIVATE NAV_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_test(NAME provider_geometry_fixture COMMAND geometry_fixture_tests)
add_executable(discovery_schedule_tests discovery_schedule_tests.cpp)
target_link_libraries(discovery_schedule_tests PRIVATE mcp_client)
add_test(NAME mcp_startup_discovery COMMAND discovery_schedule_tests)
add_executable(geometry_fallback_tests geometry_fallback_tests.cpp)
target_link_libraries(geometry_fallback_tests PRIVATE mcp_client)
target_compile_options(geometry_fallback_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(geometry_fallback_tests PRIVATE ../../firmware/components/route_parser/third_party)
target_compile_definitions(geometry_fallback_tests PRIVATE NAV_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../..")
foreach(fixture IN ITEMS complete success actual_quota transport endpoints distance steps datum polyline snap
    actual_pair_1 actual_pair_2 qps join)
    add_test(NAME geometry_fallback_${fixture} COMMAND geometry_fallback_tests ${fixture})
endforeach()
