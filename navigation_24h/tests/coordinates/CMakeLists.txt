cmake_minimum_required(VERSION 3.16)
project(coordinate_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
if(NOT TARGET nav_core)
    add_subdirectory(../../firmware/components/nav_core nav_core)
endif()
add_subdirectory(../../firmware/components/coordinates coordinates)
add_executable(coordinate_tests coordinate_tests.cpp)
target_link_libraries(coordinate_tests PRIVATE coordinates nav_core)
target_compile_options(coordinate_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME TC23_coordinate_boundary COMMAND coordinate_tests)
